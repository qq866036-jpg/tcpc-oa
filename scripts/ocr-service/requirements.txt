# 中传网 · 收文 OCR 服务依赖（OA-2463）
#
# 安装（建议独立 venv，不污染系统 python）：
#   python3 -m venv /opt/tcpc-ocr/venv
#   /opt/tcpc-ocr/venv/bin/pip install -r requirements.txt
#
# 说明：
#   · rapidocr_onnxruntime：RapidOCR 的 ONNX Runtime 发行（CPU 可跑；模型为 PP-OCR 系列中文检测/识别）。
#     新版包名 `rapidocr` 亦可（app.py 两代 API 都兼容），二者只需装一个。
#   · onnxruntime 由 rapidocr 依赖自动带入（Linux x86_64 有官方 wheel）。
#   · pypdfium2：PDF 逐页渲染（自带二进制，不依赖系统 poppler；宝塔环境无需额外装包）。
#   · pillow：图片解码（rapidocr 依赖已含，此处显式钉住下限便于审计）。
rapidocr_onnxruntime>=1.3.24
pypdfium2>=4.30.0
pillow>=10.0.0
